## [A - Shuffled Equation](https://atcoder.jp/contests/abc392/tasks/abc392_a)

???+ Abstract "题目大意"

给你三个数字，判断是否存在两个数字的乘积等于另一个数字

??? Success "参考代码"

    === "C++"
    
        ```c++
        #include <algorithm>
        #include <iostream>
    
        using namespace std;
    
        int main()
        {
            int a[3];
            cin >> a[0] >> a[1] >> a[2];
            sort(a, a + 3);
            cout << (a[0] * a[1] == a[2] ? "Yes" : "No") << endl;
            return 0;
        }
        ```

## [B - Who is Missing?](https://atcoder.jp/contests/abc392/tasks/abc392_b)

???+ Abstract "题目大意"

给你一个长度为 $M$ 的序列 $A = (A_1, A_2, ... A_M)$，序列 $A$ 的每个数字的都是 $1$ 到 $N$ 之间的数字，且序列 $A$ 的数字互不重复？问：在 $1$ 到 $N$ 中，有多少个数字没有出现在 $A$ 中？

??? Success "参考代码"

    === "C++"
    
        ```c++
        #include <iostream>
        #include <unordered_set>
    
        using namespace std;
    
        int main()
        {
            int n, m;
            cin >> n >> m;
            cout << n - m << endl;
            unordered_set<int> vis;
            while(m--)
            {
                int x;
                cin >> x;
                vis.insert(x);
            }
            for(int i = 1; i <= n; i++)
                if(!vis.count(i))
                    cout << i << ' ';
            return 0;
        }
        ```

## [C - Bib](https://atcoder.jp/contests/abc392/tasks/abc392_c)

???+ Abstract "题目大意"

有 $N$ 个人，第 $i$ 个人正在看着第 $P_i$ 个人，且第 $i$ 个人戴着号码为 $Q_i$ 的围兜。其中 $P$ 和 $Q$ 的所有数字互不重复。

对于每个 $i = 1,2,...,N$，找到戴着号码为 $i$ 的围兜的人正在看着的人身上所穿的围兜的号码。

??? Success "参考代码"

    === "C++"
    
        ```c++
        #include <iostream>
        #include <vector>
    
        using namespace std;
    
        int main()
        {
            int n;
            cin >> n;
            vector<int> p(n + 1), q(n + 1), r(n + 1);
            for (int i = 1; i <= n; i++)
                cin >> p[i];
            for (int i = 1; i <= n; i++)
            {
                cin >> q[i];
                r[q[i]] = i;
            }
            for (int i = 1; i <= n; i++)
                cout << q[p[r[i]]] << ' ';
            return 0;
        }
        ```

## [D - Doubles](https://atcoder.jp/contests/abc392/tasks/abc392_d)

???+ Abstract "题目大意"

有 $N(1 \le N \le 100)$ 个骰子，第 $i$ 个骰子有 $K_i(K_1 + K_2 + \cdots + K_N \le 10^5)$ 面，数字是 $A_{i, 1}, A_{i, 2}, ..., A_{i, k}$，每一面都是等概率的。你可以从 $N$ 个骰子中选择两个骰子，然后投掷，问：投掷得到两个相等的数字的概率最多是多少？

??? Note "解题思路"

直接枚举就好。

??? Success "参考代码"

    === "C++"
    
        ```c++
        #include <iomanip>
        #include <iostream>
        #include <unordered_map>
        #include <vector>
    
        using namespace std;
    
        int main()
        {
            int n;
            cin >> n;
            vector<unordered_map<int, double>> p(n);
            for (int i = 0; i < n; i++)
            {
                unordered_map<int, int> cnt;
                int k;
                cin >> k;
                for (int j = 0; j < k; j++)
                {
                    int x;
                    cin >> x;
                    cnt[x]++;
                }
                for (auto [x, c] : cnt)
                    p[i][x] = c / (double)k;
            }
    
            auto cal = [](auto &self, const unordered_map<int, double> &pa, const unordered_map<int, double> &pb) -> double
            {
                if (pa.size() > pb.size())
                    return self(self, pb, pa);
                double sum = 0;
                for(auto [x, p] : pa)
                {
                    auto it = pb.find(x);
                    if(it != pb.end())
                        sum += p * it->second;
                }
                return sum;
            };
    
            double ans = 0;
            for(int i = 0; i < n; i++)
                for(int j = i + 1; j < n; j++)
                    ans = max(ans, cal(cal, p[i], p[j]));
            cout << std::fixed << std::setprecision(9) << ans << endl;
            return 0;
        }
        ```

## [E - Cables and Servers](https://atcoder.jp/contests/abc392/tasks/abc392_e)

???+ Abstract "题目大意"

有一个 $N(2 \le N \le 2 \times 10 ^ 5)$ 个点，$M(N - 1 \le M \le 2 \times 10 ^ 5)$ 条边的无向图。你可以执行以下操作任意次：

-   选择任意一条边，将这条边的任意一端改为另一个点。

问：至少要操作多少次，才能让这 $N$ 个点互相都连通。输出操作方案。

??? Note "解题思路"

用并查集维护，把能合并的边都合并，不能合并的边存起来，这样会得到 $x$ 个连通分量，则操作的次数就是 $x - 1$ 次。

考虑怎么操作，因为连通只需要让两个连通分量的任意两个点连起来就好了，所以直接链接并查集中的 leader 就好，每次取出多余的一条边，选择一个 leader 尝试连接（注意这里要避开同一个集合的情况），连接上之后删掉这个 leader。可以证明这个方法一定能用最少的次数构造出连通图。 

??? Success "参考代码"

    === "C++"
    
        ```c++
        #include <algorithm>
        #include <array>
        #include <deque>
        #include <iostream>
        #include <utility>
        #include <vector>
    
        using namespace std;
    
        class DSU
        {
        public:
            explicit DSU(int n) : parent_or_size(n, -1) {}
    
            int merge(int a, int b)
            {
                int x = leader(a), y = leader(b);
                if (x == y)
                    return x;
                if (-parent_or_size[x] < -parent_or_size[y])
                    std::swap(x, y);
                parent_or_size[x] += parent_or_size[y];
                parent_or_size[y] = x;
                return x;
            }
    
            int leader(int a) { return parent_or_size[a] < 0 ? a : parent_or_size[a] = leader(parent_or_size[a]); }
    
            bool same(int a, int b) { return leader(a) == leader(b); }
    
            int size(int a) { return -parent_or_size[leader(a)]; }
    
        private:
            vector<int> parent_or_size;
        };
    
        int main()
        {
            int n, m;
            cin >> n >> m;
            DSU dsu(n + 1);
            vector<array<int, 3>> edge;
            for (int i = 1; i <= m; i++)
            {
                int u, v;
                cin >> u >> v;
                if (!dsu.same(u, v))
                    dsu.merge(u, v);
                else
                    edge.push_back({u, v, i});
            }
            deque<int> leaders;
            for (int i = 1; i <= n; i++)
                if (dsu.leader(i) == i)
                    leaders.push_back(i);
            cout << leaders.size() - 1 << '\n';
            if (leaders.size() > 1)
            {
                for (auto [u, v, id] : edge)
                {
                    if (dsu.same(u, leaders.front()))
                    {
                        v = leaders.back();
                        leaders.pop_back();
                    }
                    else
                    {
                        v = leaders.front();
                        leaders.pop_front();
                    }
                    dsu.merge(u, v);
                    cout << id << ' ' << u << ' ' << v << '\n';
                    if (leaders.size() == 1)
                        break;
                }
            }
            return 0;
        }
        ```

## [F - Insert](https://atcoder.jp/contests/abc392/tasks/abc392_f)

???+ Abstract "题目大意"

有一个空的数组 $A$。对于 $i = 1, 2, ..., N(1 \le N \le 5 \times 10 ^ 5)$，按顺序执行以下操作：

-   将数字 $i$ 插入到 $A$ 数组的第 $P_i(1 \le P_i \le i)$ 个位置。

输出所有操作结束后的数组 $A$。

??? Note "解题思路"

可以发现，越靠后（$i$ 越大）插入的数字的位置越是准确，而越前（$i$ 越小）的数字越容易被挤到后面。

如果把数字 $i$ 插入到 $P_i$ 位置，会把原本 $[P_i, i]$ 位置的所有元素都往后挪一个位置。考虑下面做法：

初始的时候，设置一个长度为 $N$ 数组 $B$，初始值全都是 $1$，$1$ 表示可以在这个位置插入数字，然后从后往前插入数字，对于 $P_i$，就在 $B$ 中找到第 $P_i$ 个 $1$，对应的位置就是数字 $i$ 的位置，然后将这个数字设置成 $0$，表示不可插入。

简单证明正确性：首先，第 $N$ 个数字插入的时候一定是对的，接着考虑 $P_{N-1}$ ，如果 $P_{N-1} < P_N$，那么 $P_{N-1}$ 不会被 $P_N$ 的影响（会插入到前面），而如果 $P_{N-1} \ge P_N$，就会被往后挤一位，这时候因为 $P_N$ 位置已经设置成 $0$ 了，所以第 $P_{N-1}$ 个 $1$ 就相应的往后挪了一位。

上述需求可以通过权值树状数组或者权值线段树实现。

??? Success "参考代码"

    === "C++"
    
        ```c++
        #include <cmath>
        #include <iostream>
        #include <vector>
    
        using namespace std;
    
        template <typename T>
        class FenwickTree
        {
        public:
            FenwickTree(int n) : n(n), t(n)
            {
                for(int i = 1; i < n; i++)
                {
                    t[i]++;
                    int j = i + lowbit(i);
                    if(j < n)
                        t[j] += t[i];
                }
            }
    
            void add(int x, T k)
            {
                for (; x < n; x += lowbit(x))
                    t[x] += k;
            }
    
            int lower_bound(T k) const
            {
                int x = 0;
                T sum = 0;
                for (int i = 1 << (int)log2(n); i; i >>= 1)
                {
                    if (x + i >= n || sum + t[x + i] >= k)
                        continue;
                    x += i;
                    sum += t[x];
                }
                return x + 1;
            }
    
            T range(int l, int r) const { return prefix(r) - prefix(l - 1); }
    
        private:
            int n;
            vector<T> t;
    
            int lowbit(int x) const { return x & -x; }
    
            T prefix(int x) const
            {
                T ans = 0;
                for (; x; x -= lowbit(x))
                    ans += t[x];
                return ans;
            }
        };
    
        int main()
        {
            int n;
            cin >> n;
            vector<int> p(n + 1);
            for (int i = 1; i <= n; i++)
                cin >> p[i];
            FenwickTree<int> t(n + 1);
            vector<int> ans(n + 1);
            for (int i = n; i; i--)
            {
                int x = t.lower_bound(p[i]);
                ans[x] = i;
                t.add(x, -1);
            }
            for (int i = 1; i <= n; i++)
                cout << ans[i] << ' ';
            return 0;
        }
        ```

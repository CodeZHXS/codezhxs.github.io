## [B - Grid Rotation](https://atcoder.jp/contests/abc404/tasks/abc404_b)

???+ Abstract "题目大意"

给你两个 $N \times N(1 \le N \le 100)$ 的二维网格  $S$ 和 $T$。

$S$ 和 $T$ 的每个格子都被染成了黑白两种颜色中的一种，其中字符 `#` 代表黑色，字符 `.` 代表白色。
  
你可以执行以下两种操作任意次：
- 选择一个 $S$ 中的格子，如果它是白色的，将它染成黑色；如果它是黑色的，将它染成白色。
- 将 $S$ 顺时针旋转 $90°$。

问：最少需要操作多少次才能使得 $S$ 和 $T$ 完全相同？

??? Success "参考代码"

    === "C++"
    
        ```c++
        #include <iostream>
        #include <limits>
        #include <string>
        #include <vector>

        using namespace std;

        int main()
        {
            int n;
            cin >> n;
            vector<string> s(n), t(n);
            for (int i = 0; i < n; i++)
                cin >> s[i];
            for (int i = 0; i < n; i++)
                cin >> t[i];

            auto rotate = [n](const vector<string> &s) -> vector<string>
            {
                vector<string> ans = s;
                for (int i = 0; i < n; i++)
                    for (int j = 0; j < n; j++)
                        ans[j][n - i - 1] = s[i][j];
                return ans;
            };

            auto cal = [n](int cnt, const vector<string> &s, const vector<string> &t) -> int
            {
                for (int i = 0; i < n; i++)
                    for (int j = 0; j < n; j++)
                        cnt += s[i][j] != t[i][j];
                return cnt;
            };

            int ans = numeric_limits<int>::max();
            for (int i = 0; i < 4; i++)
            {
                ans = min(ans, cal(i, s, t));
                s = rotate(s);
            }
            cout << ans << endl;

            return 0;
        }
        ```

## [C - Cycle Graph?](https://atcoder.jp/contests/abc404/tasks/abc404_c)

???+ Abstract "题目大意"

给你一个 $N(3 \le N \le 2 \times 10 ^ 5)$ 个点 $M(0 \le M 2 \times 10 ^ 5)$ 条边的简单无向图。

判断这个图是否是“环图”。一个图是“环图”当且仅当图中存在一个环包含图中的所有点，并且图上没有其他的边。

??? Note "解题思路"

可以维护一个并查集，同时算出每个点的度数，那么“环图”的充要条件就是：
- $N = M$，即边数等于点数
- 图连通分量数为 $1$，即图是连通的。
- 每个点的度数都是 $2$。

??? Success "参考代码"

    === "C++"
    
        ```c++
        #include <iostream>
        #include <utility>
        #include <vector>

        using namespace std;

        class DSU
        {
        public:
            explicit DSU(int n) : parent_or_size(n, -1) {}

            int merge(int a, int b)
            {
                int x = leader(a), y = leader(b);
                if (x == y)
                    return x;
                if (-parent_or_size[x] < -parent_or_size[y])
                    swap(x, y);
                parent_or_size[x] += parent_or_size[y];
                parent_or_size[y] = x;
                return x;
            }

            int leader(int a) { return parent_or_size[a] < 0 ? a : parent_or_size[a] = leader(parent_or_size[a]); }

            bool same(int a, int b) { return leader(a) == leader(b); }

            int size(int a) { return -parent_or_size[leader(a)]; }

        private:
            vector<int> parent_or_size;
        };

        int main()
        {
            int n, m;
            cin >> n >> m;
            vector<int> deg(n + 1);
            DSU dsu(n + 1);
            for (int i = 0; i < m; i++)
            {
                int u, v;
                cin >> u >> v;
                deg[u]++, deg[v]++;
                if (!dsu.same(u, v))
                    dsu.merge(u, v);
            }

            auto check = [n, m, &deg, &dsu]() -> bool
            {
                if (n != m)
                    return false;
                if (dsu.size(1) != n)
                    return false;
                for (int i = 1; i <= n; i++)
                    if (deg[i] != 2)
                        return false;
                return true;
            };

            cout << (check() ? "Yes" : "No") << endl;
            return 0;
        }
        ```

## [D - Goin' to the Zoo](https://atcoder.jp/contests/abc404/tasks/abc404_d)

???+ Abstract "题目大意"

有 $N(1 \le N \le 10)$ 个动物园，编号从 1 到 $N$，第 $i$ 个动物园的门票费用是 $C_i$ 元。

有 $M(1 \le M \le 100)$ 种动物，编号从 1 到 $M$。

第 $i$ 种动物在 $K_i(1 \le K_i \le N)$ 个不同的动物园中被展出，分别是第 $A_{i,1}, A_{i,2}, \ldots, A_{i,K_i}(1 \le A_{i, j} \le N)$ 个动物园。

你打算去游览一些动物园，并且每一种动物你都打算观赏至少两次。你可以游览同一个动物园多次，如果游览了同一个动物园多次，我们认为动物园中的动物也被观赏了多次。

问：为了让所有动物都被观赏至少两次，至少需要花费多少钱？

??? Note "解题思路"

$N \le 10$, 且一个动物园只可能游览 $0$ 到 $2$ 次，直接搜索就好，时间复杂度 $O(3^N)$。

??? Success "参考代码"

    === "C++"
    
        ```c++
        #include <cmath>
        #include <iostream>
        #include <limits>
        #include <vector>

        using namespace std;
        using LL = long long;

        int main()
        {
            int n, m;
            cin >> n >> m;
            vector<int> c(n);
            for (int i = 0; i < n; i++)
                cin >> c[i];
            vector<vector<int>> ani(n);
            for (int i = 0; i < m; i++)
            {
                int k;
                cin >> k;
                while (k--)
                {
                    int a;
                    cin >> a;
                    a--;
                    ani[a].push_back(i);
                }
            }

            LL ans = numeric_limits<LL>::max();
            vector<int> cnt(m, 0);

            auto check = [&cnt]() -> bool
            {
                for (auto x : cnt)
                    if (x < 2)
                        return false;
                return true;
            };
            
            auto dfs = [n, &ani, &cnt, &c, &check, &ans](auto &self, int cur, LL fee) -> void
            {
                if (cur == n)
                {
                    if (check())
                        ans = min(ans, fee);
                    return;
                }

                self(self, cur + 1, fee);

                for (auto x : ani[cur])
                    cnt[x]++;
                self(self, cur + 1, fee + c[cur]);

                for (auto x : ani[cur])
                    cnt[x]++;
                self(self, cur + 1, fee + 2 * c[cur]);

                for (auto x : ani[cur])
                    cnt[x] -= 2;
            };

            dfs(dfs, 0, 0);
            cout << ans << endl;
            return 0;
        }
        ```

???+ Abstract "题目大意"

有 $N(1 \le 2000)$ 个碗排成一排，从左往右依次编号 $0,1,...,N−1$。

除了第 $0$ 个碗，其他碗上面都写有一个数字 $C_i$，且初始的时候第 $i$ 个碗中装有 $A_i$ 颗豆子。第 $0$ 个碗上没有数字，且里面没有豆子。

你可以执行以下操作任意次：

- 选择一个有豆子的碗 $i$，从其中拿走部分豆子（可以全拿）。
- 将拿走的这些豆子任意分配到 $i - C_i, i - C_i + 1, ... i - 1$ 这些碗当中。
- 
问：至少要操作几次，才能将所有的豆子都放入到第 $0$ 个碗中？

??? Note "解题思路"

简单贪心一下会发现，最优解一定是拿走第 $i$ 个碗的所有豆子，然后将其全部放到另一个碗。

下面考虑怎么选出最优的碗。显然要从后往前遍历，假设当前碗是 $i$，如果 $[i - C_i, i - 1]$ 中至少存在一个碗 有豆子，那么直接把这些豆子全部放到这个有豆子的碗一起转移，这样代价为 $1$。而尽可能将豆子合并到一个碗中是最优的。

如果不存在这样的碗，那么对于所有的 $i - C_i \le j \le i - 1$，分别求出 $j - C_j$ 然后取最小值，向最小的 $j - C_j$ 转移，这是因为更小的 $j - C_j$ 能移动的更广，更容易转移到有豆子的碗。

实现的时候要特判一下，如果 $i - C_i \le 0$，那直接答案加 $1$ 就好，不然会 WA 几个点。

??? Success "参考代码"

    === "C++"
    
        ```c++
        #include <iostream>
        #include <vector>

        using namespace std;

        int main()
        {
            int n;
            cin >> n;
            vector<int> a(n), c(n);
            for (int i = 1; i < n; i++)
                cin >> c[i];
            for (int i = 1; i < n; i++)
                cin >> a[i];
            int ans = 0;
            for (int i = n - 1; i; i--)
            {
                if (!a[i])
                    continue;
                ans++;
                if (i - c[i] <= 0)
                    continue;
                int pos = i - 1;
                for (int j = i - c[i]; j < i; j++)
                {
                    if (a[j])
                    {
                        pos = j;
                        break;
                    }
                    if (j - c[j] < pos - c[pos])
                        pos = j;
                }
                a[pos] += a[i];
            }
            cout << ans << endl;
            return 0;
        }
        ```

## [F - Lost and Pound](https://atcoder.jp/contests/abc404/tasks/abc404_f)

???+ Abstract "题目大意"

青木和高桥在玩一个游戏。场上有 $N(1 \le N \le 2 \times 10^5)$ 个按钮，其中 $1$ 个是关键按钮，剩下的 $N−1$ 个是普通按钮。

青木知道哪一个按钮是关键按钮，而高桥并不知道，也就是说，高桥无法将这 $N$ 个按钮相互区分。

在游戏中，以下事件将会被重复 $T(1 \le T \le 30)$ 次：

1. 青木将这 $N$ 个按钮按随机的顺序摆放。
2. 高桥进行 $M(1 \le M \le 30)$ 次操作，每次操作中他将选择一个按钮并按下它一次。在这个过程中他可以选择按下同一个按钮多次。
3. 青木告诉高桥从游戏开始至今，关键按钮被高桥按下了多少次。

如果在这 $T$ 轮过程中被高桥按下了关键按钮至少 $K(1 \le K \le 30)$ 次，则高桥获胜。请你求出当高桥以最优策略游玩游戏时，他获胜的概率，并以浮点数的形式输出。

??? Note "解题思路"

定义 $dp[i][j]$ 表示还剩下 $i$ 轮，且至少需要按下 $j$ 次按钮才能获胜的概率。下面考虑怎么转移。不妨假设一共按下了 $k$ 个不同的按钮，其中第一个按钮按了 $a_1$ 次，第二个按钮按了 $a_2$ 次，...，第 $k$ 个按钮按了 $a_k$ 次。显然 $a_1 + a_2 + ... + a_k = M$，而且 $a_i \ge 1$，这里如果直接钦定 $a_i \ge a_{i + 1}$, 那枚举起来就会容易很多（数量等于划分数），随便剪一下枝就能过。

??? Success "参考代码"

    === "C++"
    
        ```c++
        #include <algorithm>
        #include <iomanip>
        #include <iostream>
        #include <vector>

        using namespace std;

        int main()
        {
            int n, t, m, k;
            cin >> n >> t >> m >> k;
            vector<vector<double>> dp(t + 1, vector<double>(k + 1));
            dp[0][0] = 1;
            for (int tt = 1; tt <= t; tt++)
                for (int kk = 0; kk <= k; kk++)
                {
                    auto dfs = [n, tt, kk, &dp](auto &&self, int i, int last, int left, double val) -> void
                    {
                        if (left == 0)
                        {
                            val += (n - i) * dp[tt - 1][kk];
                            val /= n;
                            dp[tt][kk] = max(dp[tt][kk], val);
                            return;
                        }
                        for (int x = min(last, left); x; x--)
                        {
                            if ((n - i) * x < left)
                                break;
                            self(self, i + 1, x, left - x, val + dp[tt - 1][max(kk - x, 0)]);
                        }
                    };

                    dfs(dfs, 0, m, m, 0);
                }
            cout << setprecision(10) << dp[t][k] << endl;
            return 0;
        }
        ```

## [G - Specified Range Sums](https://atcoder.jp/contests/abc404/tasks/abc404_g)

???+ Abstract "题目大意"

给你一个整数 $N(1 \le N \le 4000)$ 和 $M(1 \le M \le 4000)$ 个约束条件，每一个约束条件由 $(L_i, R_i, S_i)$ 组成，其中 $1 \le L_i \le R_i \le N$，$1 \le S_i \le 10^9$。

问：是否存在一个长度为 $N$ 的**正整数**序列 $A$ 满足所有的约束条件？约束条件 $(L_i, R_i, S_i)$ 的含义是：
- $\sum\limits_{j=L_i}^{R_i} A_j = S_i$

如果存在多个满足条件的序列 $A$，输出 $A$ 的所有数字之和最小的那一个。如果不存在满足条件的序列 $A$，输出 $-1$。

??? Note "解题思路"

首先将约束条件转成前缀和 $P_{R_i} - P_{L_i - 1} = S_i$，这是一个等式，转换成 $P_{R_i} - P_{L_i - 1} \le S_i$ 和 $P_{R_i} - P_{L_i - 1} \ge S_i$，然后就可以用差分约束建图了。注意 $A_i$ 都是正整数，也就是 $S_i - S_{i - 1} \ge 1$。

然后跑最短路就好。注意前缀和的 $P_0$ 是一定为 $0$ 的，所以答案实际是 $P_N - P_0$，相当于在原来的基础上增加了 $-P_0$

??? Success "参考代码"

    === "C++"
    
        ```c++
        #include <iostream>
        #include <limits>
        #include <queue>
        #include <utility>
        #include <vector>

        using namespace std;
        using LL = long long;

        struct Graph
        {
            struct Edge
            {
                int to, cost;
            };

            int n;
            vector<vector<Edge>> edge;

            Graph(int n) : n(n), edge(n) {}

            void add(int u, int v, int w) { edge[u].push_back({v, w}); }

            pair<bool, vector<LL>> spfa(int s)
            {
                vector<LL> dis(n, numeric_limits<LL>::max());
                vector<int> cnt(n, 0);
                dis[s] = 0;
                queue<int> q;
                vector<bool> inq(n);
                q.push(s);
                inq[s] = true;
                while (!q.empty())
                {
                    int u = q.front();
                    q.pop();
                    inq[u] = false;
                    for (auto [v, w] : edge[u])
                    {
                        if (dis[v] > dis[u] + w)
                        {
                            dis[v] = dis[u] + w;
                            cnt[v] = cnt[u] + 1;
                            if (cnt[v] > n)
                                return {false, {}};
                            if (!inq[v])
                            {
                                q.push(v);
                                inq[v] = true;
                            }
                        }
                    }
                }
                return {true, dis};
            }
        };

        int main()
        {
            int n, m;
            cin >> n >> m;
            int s = n + 1;
            Graph g(n + 2);
            while (m--)
            {
                int l, r, s;
                cin >> l >> r >> s;
                g.add(l - 1, r, s);
                g.add(r, l - 1, -s);
            }
            for (int i = 1; i <= n; i++)
                g.add(i, i - 1, -1);
            for (int i = 0; i <= n; i++)
                g.add(s, i, 0);
            auto [ok, dis] = g.spfa(s);
            if (!ok)
            {
                cout << -1 << endl;
                return 0;
            }
            cout << dis[n] - dis[0] << endl;
            return 0;
        }
        ```
